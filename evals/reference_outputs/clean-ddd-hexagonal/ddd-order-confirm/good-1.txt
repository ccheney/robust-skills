export class Order{constructor(id,items){if(items.some(i=>!Number.isSafeInteger(i.quantity)||i.quantity<=0||!Number.isSafeInteger(i.priceCents)||i.priceCents<0))throw Error("item");this.id=id;this.items=items.map(i=>({...i}));this.status="DRAFT";this.events=[];}get totalCents(){return this.items.reduce((sum,i)=>sum+i.quantity*i.priceCents,0);}confirm(){if(!this.items.length||this.status!=="DRAFT")throw Error("state");this.status="CONFIRMED";this.events.push({type:"OrderConfirmed",orderId:this.id});}}
